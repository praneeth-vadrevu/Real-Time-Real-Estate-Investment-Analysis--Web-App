147 // ---------- Amortization helper (annual) ----------
148 static class Amort {
149 final double P; // principal
150 final double r; // APR (as fraction)
151 final int nYears; // total term (years)
152 final int ioYears; // initial interest-only years
153
154 Amort(double principal, double rate, int years, int ioYears) {
155 this.P = principal; this.r = rate; this.nYears = years; this.ioYears = Math.min(ioYears, years);
156 }
157
159 double annualDebtService(int year) {
160 if (P <= 0) return 0.0;
161 if (year <= ioYears) return P * r;
162 int remaining = nYears - ioYears;
163 if (r == 0) return P / remaining;
164 double a = (P * r) / (1 - Math.pow(1 + r, -remaining));
165 return a;
166 }
167
169 double balanceEndOfYear(int year) {
170 if (P <= 0) return 0.0;
171 if (year <= ioYears) return P;
172 int k = year - ioYears; int remaining = nYears - ioYears;
173 if (r == 0) return Math.max(0, P - (P / remaining) * k);
174 double a = (P * r) / (1 - Math.pow(1 + r, -remaining));
175 double bal = P * Math.pow(1 + r, k) - a * ((Math.pow(1 + r, k) - 1) / r);
176 return Math.max(0, bal);
177 }
178
180 double principalPaidYear(int year) {
181 if (year <= ioYears || P <= 0) return 0.0;
182 double start = balanceEndOfYear(year - 1);
183 double end = balanceEndOfYear(year);
184 return Math.max(0, start - end);
185 }
186 }

195 private static Double irr(List<Double> cash, double guess) {
196 double x = guess;
197 for (int i = 0; i < 50; i++) {
198 double f = 0, df = 0;
199 for (int t = 0; t < cash.size(); t++) {
200 double ct = cash.get(t), d = Math.pow(1 + x, t);
201 f += ct / d;
202 if (t > 0) df += -t * ct / Math.pow(1 + x, t + 1);
203 }
204 double x1 = x - f / df;
205 if (Double.isNaN(x1) || Double.isInfinite(x1)) break;
206 if (Math.abs(x1 - x) < 1e-7) return x1;
207 x = x1;
208 }
209 return null;
