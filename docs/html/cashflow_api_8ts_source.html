184export async function analyzeCashflow(request: CashflowRequest): Promise<CashflowResponse> {
185 try {
186 const response = await fetch('http://localhost:8080/api/analysis/cashflow', {
187 method: 'POST',
188 headers: {
189 'Content-Type': 'application/json',
190 },
191 body: JSON.stringify(request),
192 });
193
194 if (!response.ok) {
195 if (response.status === 0 || response.status === 503 || response.status === 502) {
196 throw new Error('BACKEND_NOT_RUNNING');
197 }
198 const errorText = await response.text();
199 throw new Error(`Failed to analyze cashflow: ${response.status} ${response.statusText}. ${errorText}`);
200 }
201
202 const result = await response.json();
203 return result;
204 } catch (error: any) {
205 console.error('Error calling cashflow API:', error);
206 if (error.message === 'BACKEND_NOT_RUNNING' || error.message?.includes('Failed to fetch') || error.name === 'TypeError') {
207 throw new Error('Backend server is not running. Please start the backend server on port 8080.');
208 }
209 throw error;
210 }
