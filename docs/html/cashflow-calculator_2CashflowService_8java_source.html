267 static class Amort {
268 final double principal; // Loan principal amount
269 final double annualRate; // Annual interest rate as fraction (e.g., 0.0675 for 6.75%)
270 final int totalYears; // Total amortization term in years
271 final int interestOnlyYears; // Number of initial years that are interest-only

281 Amort(double principal, double rate, int years, int ioYears) {
282 this.principal = principal;
283 this.annualRate = rate;
284 this.totalYears = years;
285 this.interestOnlyYears = Math.min(ioYears, years);
286 }

295 double annualDebtService(int year) {
296 if (principal <= 0) return 0.0;
297
298 // During interest-only period, payment is just interest
299 if (year <= interestOnlyYears) return principal * annualRate;
300
301 // Calculate amortizing payment for remaining term
302 int remainingYears = totalYears - interestOnlyYears;
303 if (annualRate == 0) return principal / remainingYears;
304
305 // Standard amortization formula: P * r / (1 - (1+r)^-n)
306 double payment = (principal * annualRate) / (1 - Math.pow(1 + annualRate, -remainingYears));
307 return payment;
308 }
309
316 double balanceEndOfYear(int year) {
317 if (principal <= 0) return 0.0;
318
319 // During interest-only period, balance remains at principal
320 if (year <= interestOnlyYears) return principal;
321
322 // Calculate balance after interest-only period
323 int yearsIntoAmortization = year - interestOnlyYears;
324 int remainingYears = totalYears - interestOnlyYears;
325
326 if (annualRate == 0) {
327 // Simple linear paydown for zero interest
328 return Math.max(0, principal - (principal / remainingYears) * yearsIntoAmortization);
329 }
330
331 // Standard amortization balance formula
332 double payment = (principal * annualRate) / (1 - Math.pow(1 + annualRate, -remainingYears));
333 double balance = principal * Math.pow(1 + annualRate, yearsIntoAmortization)
334 - payment * ((Math.pow(1 + annualRate, yearsIntoAmortization) - 1) / annualRate);
335 return Math.max(0, balance);
336 }
337
345 double principalPaidYear(int year) {
346 if (year <= interestOnlyYears || principal <= 0) return 0.0;
347
348 double balanceAtStart = balanceEndOfYear(year - 1);
349 double balanceAtEnd = balanceEndOfYear(year);
350 return Math.max(0, balanceAtStart - balanceAtEnd);
351 }

402 private static Double irr(List<Double> cashFlows, double initialGuess) {
403 double currentRate = initialGuess;
404
405 // Newton-Raphson iteration (max 50 iterations)
406 for (int iteration = 0; iteration < 50; iteration++) {
407 double netPresentValue = 0;
408 double netPresentValueDerivative = 0;
409
410 // Calculate NPV and its derivative
411 for (int period = 0; period < cashFlows.size(); period++) {
412 double cashFlow = cashFlows.get(period);
413 double discountFactor = Math.pow(1 + currentRate, period);
414 netPresentValue += cashFlow / discountFactor;
415
416 // Derivative calculation (skip period 0 as it has no time component)
417 if (period > 0) {
418 netPresentValueDerivative += -period * cashFlow / Math.pow(1 + currentRate, period + 1);
419 }
420 }
421
422 // Newton-Raphson step: x_new = x_old - f(x) / f'(x)
423 double newRate = currentRate - netPresentValue / netPresentValueDerivative;
424
425 // Check for invalid values
426 if (Double.isNaN(newRate) || Double.isInfinite(newRate)) {
427 break;
428 }
429
430 // Check for convergence
431 if (Math.abs(newRate - currentRate) < 1e-7) {
432 return newRate;
433 }
434
435 currentRate = newRate;
436 }
437
438 // Convergence failed
439 return null;
440 }
