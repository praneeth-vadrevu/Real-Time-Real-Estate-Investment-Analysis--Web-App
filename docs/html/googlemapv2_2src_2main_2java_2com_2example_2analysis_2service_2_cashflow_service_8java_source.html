130 static class Amort {
131 final double P;
132 final double r;
133 final int nYears;
134 final int ioYears;
135
136 Amort(double principal, double rate, int years, int ioYears) {
137 this.P = principal; this.r = rate; this.nYears = years; this.ioYears = Math.min(ioYears, years);
138 }
139
140 double annualDebtService(int year) {
141 if (P <= 0) return 0.0;
142 if (year <= ioYears) return P * r;
143 int remaining = nYears - ioYears;
144 if (r == 0) return P / remaining;
145 double a = (P * r) / (1 - Math.pow(1 + r, -remaining));
146 return a;
147 }
148
149 double balanceEndOfYear(int year) {
150 if (P <= 0) return 0.0;
151 if (year <= ioYears) return P;
152 int k = year - ioYears; int remaining = nYears - ioYears;
153 if (r == 0) return Math.max(0, P - (P / remaining) * k);
154 double a = (P * r) / (1 - Math.pow(1 + r, -remaining));
155 double bal = P * Math.pow(1 + r, k) - a * ((Math.pow(1 + r, k) - 1) / r);
156 return Math.max(0, bal);
157 }
158
159 double principalPaidYear(int year) {
160 if (year <= ioYears || P <= 0) return 0.0;
161 double start = balanceEndOfYear(year - 1);
162 double end = balanceEndOfYear(year);
163 return Math.max(0, start - end);
164 }
